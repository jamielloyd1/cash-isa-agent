# Example 1: Check if a user is eligible for a Cash ISA
def is_user_eligible(user, policy):
    """
    Determine if a user is eligible for a Cash ISA.

    Args:
        user (User): User model object
        policy (CashISAPolicy): Policy model object

    Returns:
        bool: True if eligible, False otherwise
    """
    if user.age < policy.minimum_age:
        return False
    if user.cash_isa_contributions_ytd >= policy.annual_allowance:
        return False
    return True


# Example 2: Calculate the remaining allowance a user can put in a Cash ISA
def remaining_allowance(user, policy):
    """
    Calculate how much more a user can contribute to their Cash ISA.

    Args:
        user (User): User model object
        policy (CashISAPolicy): Policy model object

    Returns:
        float: Remaining allowance (>=0)
    """
    remaining = policy.annual_allowance - user.cash_isa_contributions_ytd
    return max(remaining, 0)


# Example 3: Determine if a user can open a new Cash ISA
def can_open_new_isa(user, policy):
    """
    Determine if the user can open a new Cash ISA account.

    Args:
        user (User): User model object
        policy (CashISAPolicy): Policy model object

    Returns:
        bool: True if user can open a new ISA, False otherwise
    """
    if user.existing_cash_isa:
        return False
    if user.age < policy.minimum_age:
        return False
    return True


# Example 4: Check if user contribution is close to limit
def is_near_allowance_limit(user, policy, threshold_percent=90):
    """
    Check if the user has contributed close to their annual allowance.

    Args:
        user (User): User model object
        policy (CashISAPolicy): Policy model object
        threshold_percent (int): Percentage of allowance considered "near limit"

    Returns:
        bool: True if user's contributions are >= threshold_percent of allowance
    """
    percentage = (user.cash_isa_contributions_ytd / policy.annual_allowance) * 100
    return percentage >= threshold_percent
